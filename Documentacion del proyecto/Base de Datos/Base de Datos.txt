CREATE DATABASE auto1;
USE auto1;

CREATE TABLE auto (
    clave INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL,
    ap1 VARCHAR(30) NOT NULL,
    ap2 VARCHAR(30) NOT NULL,
    modelo VARCHAR(30) NOT NULL,
    precio FLOAT NOT NULL, 
    enganche FLOAT NOT NULL,
    pagomensual FLOAT NOT NULL,
    meses INT NOT NULL
);
