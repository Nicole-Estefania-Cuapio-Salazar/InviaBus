<?php
$conn = new mysqli("localhost", "root", "", "empresa");

if ($conn->connect_error) {
    die("Conexión fallida: " . $conn->connect_error);
}

$accion = $_POST['accion'];

switch ($accion) {
    case 'consultar': //se puede ocupar este codigo solo es de ver como implementarlo en lo que es la conexión la base de datos 
        $matricula = $_POST['matricula'];
        $stmt = $conn->prepare("SELECT * FROM empleados WHERE id=?");
        $stmt->bind_param("s", $matricula);
        $stmt->execute();
        $res = $stmt->get_result();

        if ($res->num_rows > 0) {
            $row = $res->fetch_assoc();
            echo "Nombre: {$row['nombre']}<br>Puesto: {$row['puesto']}<br>Salario: {$row['salario']}";
        } else {
            echo "Empleado no encontrado";
        }
        break;

    case 'actualizar':
        $matricula = $_POST['matricula'];
        $nombre = $_POST['nombre'];
        $puesto = $_POST['puesto'];
        $salario = $_POST['salario'];

        $stmt = $conn->prepare("UPDATE empleados SET nombre=?, puesto=?, salario=? WHERE matricula=?");
        $stmt->bind_param("ssds", $nombre, $puesto, $salario, $matricula);
        $stmt->execute();

        echo "Datos actualizados correctamente";
        break;

    case 'insertar':
        $matricula = $_POST['matricula'];
        $password = $_POST['password'];
        $nombre = $_POST['nombre'];
        $puesto = $_POST['puesto'];
        $salario = $_POST['salario'];

        // Verificar si ya existe la matrícula
        $stmt = $conn->prepare("SELECT * FROM empleados WHERE matricula=?");
        $stmt->bind_param("s", $matricula);
        $stmt->execute();
        $res = $stmt->get_result();

        if ($res->num_rows > 0) {
            echo "Error: La matrícula ya está registrada.";
        } else {
            $stmt = $conn->prepare("INSERT INTO empleados (matricula, password, nombre, puesto, salario) VALUES (?, ?, ?, ?, ?)");
            $stmt->bind_param("ssssd", $matricula, $password, $nombre, $puesto, $salario);
            $stmt->execute();
            echo "Empleado registrado correctamente.";
        }
        break;

    case 'borrar':
        $matricula = $_POST['matricula'];
        $stmt = $conn->prepare("DELETE FROM empleados WHERE matricula=?");
        $stmt->bind_param("s", $matricula);
        $stmt->execute();
        echo "Empleado eliminado correctamente.";
        break;

    default:
        echo "Acción no válida.";
}

$conn->close();
?>
